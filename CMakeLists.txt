cmake_minimum_required(VERSION 3.5)
project(APKPatcher1018 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(NearBranch OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/Dobby)
add_library(backend1018 SHARED native/backend.cpp)
target_include_directories(backend1018 PRIVATE vendor/Dobby/include)
target_link_libraries(backend1018 PRIVATE dobby log dl)
target_compile_options(backend1018 PRIVATE -Wall -Wextra -fvisibility=hidden)
set_target_properties(backend1018 PROPERTIES LINK_FLAGS "-Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL")
